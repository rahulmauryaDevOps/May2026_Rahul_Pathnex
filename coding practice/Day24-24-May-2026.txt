3. Kubernetes — Node Affinity
====================================================

apiVersion: v1
kind: Pod

metadata:
  name: pathnex-affinity

spec:
  affinity:
    nodeAffinity:
      requiredDuringSchedulingIgnoredDuringExecution:
        nodeSelectorTerms:
          - matchExpressions:
              - key: disktype
                operator: In
                values:
                  - ssd

  containers:
    - name: web
      image: nginx

====================================================
5. Jenkins Pipeline — Rollback on Failure
====================================================

pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
    }

    stages {
        stage('Deploy') {
            steps {
                script {
                    try {
                        sh 'echo "Deploying $INSTITUTE_NAME application..."'
                        sh 'exit 1'
                    } catch (Exception e) {
                        echo "Deployment failed. Rolling back $INSTITUTE_NAME application..."
                    }
                }
            }
        }
    }
}

6. GitLab CI — Rollback Simulation
====================================================

stages:
  - deploy

variables:
  INSTITUTE_NAME: "Pathnex"

deploy:
  stage: deploy
  image: alpine:latest

  script:
    - echo "Deploying $INSTITUTE_NAME application..."
    - exit 1

  after_script:
    - echo "Deployment failed. Rolling back $INSTITUTE_NAME application..."
