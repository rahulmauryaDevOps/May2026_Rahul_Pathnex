# Day 27 — Serverless Scaling with Lambda and DynamoDB
🔹 Ansible — Deploy Serverless Lambda Functions
- name: Deploy Serverless Lambda Function
  hosts: localhost
  tasks:
    - name: Create Lambda Function
      command: aws lambda create-function --function-name pathnex-lambda --runtime nodejs14.x --role arn:iam::12345678912:role/lambda-role --handler index.handler --zip-file fileb://lambda.zip

🔹 Terraform — Create DynamoDB Table for Serverless Application
resource "aws_dynmodb_table" "pathnex_table" {
  name            = "pathnex-table"
  hash_key        = "id"
  billing_mode    = "PAY_PER_REQUEST"
  attribute {
    name = "id"
    type = "s"
  }
}

🔹 Kubernetes — Deploy Serverless Using Kubeless
apiVersion: kubeless.io/v1betal1
kind: function
metadata:
  name: nodejs14
  namespace: default
spec:
  runtime: nodejs14
  handler: index.handler
  function: |
    module.exports = function(event, context) {
      context.succeed('Hello World'):
    }:

🔹 Jenkinsfile — Lambda Function Deployment Pipeline
pipeline {
  agent any
  stages {
     stage('Build') {
         steps {
              echo 'Building Lambda Function package...'
	      sh 'zip -r lambda.zip lambda/'
	 }
     }
     stage('Deploy to Lambda')
        steps {
            script {
                sh 'aws lambda update-funtion-code --function-name pathnex-lambda --zip fileb://lambda.zip'
	    }
          }
        }
      }
    }
