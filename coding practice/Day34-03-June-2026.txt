4. Jenkins Pipeline — Automatic Rollback
====================================================

pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM = "DevOps"
        ENV = "prod"
    }

    stages {

        stage('Deploy') {

            steps {

                script {

                    try {

                        sh 'echo "Deploying $INSTITUTE_NAME app in $ENV for $TEAM"'

                        sh 'exit 1'

                    } catch (Exception e) {

                        echo "Deployment failed. Rolling back $INSTITUTE_NAME app in $ENV"
                    }
                }
            }
        }
    }
}

Commands Practiced:

mkdir jenkins-day28
cd jenkins-day28

vim Jenkinsfile

Concept:
- Rollback Strategy
- try-catch
- Deployment Failure Handling

====================================================
5. GitLab CI — Rollback Strategy
====================================================

stages:
  - deploy

variables:
  INSTITUTE_NAME: "Pathnex"
  TEAM: "DevOps"
  ENV: "prod"

deploy:
  stage: deploy

  image: alpine:latest

  script:
    - echo "Deploying $INSTITUTE_NAME app in $ENV for $TEAM"
    - exit 1

  after_script:
    - echo "Deployment failed. Rolling back $INSTITUTE_NAME app in $ENV"

Commands Practiced:

mkdir gitlab-day28
cd gitlab-day28

vim .gitlab-ci.yml

Concept:
- Rollback Strategy
- Deployment Failure Handling

- GitLab CI
