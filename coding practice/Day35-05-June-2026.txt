1. Ansible — Create User + SSH Key + Shell
====================================================

---
- name: Create Pathnex Admin User
  hosts: all
  become: yes

  tasks:

    - name: Create user
      user:
        name: pathnex-admin
        shell: /bin/bash

    - name: Add SSH Key
      authorized_key:
        user: pathnex-admin
        state: present
        key: "ssh-rsa AAAAB3NzaC1yc2EAAAADAQABAAABAQCPathnexKey"

3. Kubernetes — Deployment + ClusterIP
====================================================

apiVersion: apps/v1
kind: Deployment

metadata:
  name: pathnex-deploy

spec:
  replicas: 2

  selector:
    matchLabels:
      app: pathnex

  template:
    metadata:
      labels:
        app: pathnex

    spec:
      containers:
        - name: nginx
          image: nginx

---

apiVersion: v1
kind: Service

metadata:
  name: pathnex-clusterip

spec:
  type: ClusterIP

4. Shell Script — Menu Driven Monitoring
====================================================

#!/bin/bash

echo "1) IP"
echo "2) Disk"
echo "3) Memory"

read -p "Choose option: " opt

case $opt in

1)
hostname -I
;;

2)
df -h
;;

3)
free -h
;;

*)
echo "Invalid Choice"
;;

esac

Commands Practiced:

mkdir shell-day30

5. Jenkins Pipeline Mini Project
====================================================

pipeline {

    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM = "DevOps"
        ENV = "prod"
        PROJECT = "Pathnex-Training"
    }

    stages {

        stage('Infrastructure') {

            steps {

                sh '''
                terraform init
                terraform validate
                terraform plan
                '''
            }
        }

        stage('App Deployment') {

            steps {

                sh '''
                ansible-playbook user.yml

                kubectl apply -f deployment.yml
                '''
            }
        }

        stage('Monitoring Checks') {

            steps {

                sh '''
                echo "===== IP ====="
                hostname -I

                echo "===== DISK ====="
                df -h

                echo "===== MEMORY ====="
                free -h
                '''
            }
        }
    }
}
